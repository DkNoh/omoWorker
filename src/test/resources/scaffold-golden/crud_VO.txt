 */
@Data
public class SmsHistoryVO {

    /** SELECT 결과 컬럼에서 생성된 조회 필드. */
    private LocalDate sendDt;
    /** SELECT 결과 컬럼에서 생성된 조회 필드. */
    private String receiverNo;
}
